cmake_minimum_required(VERSION 3.20)
project(ndp_library LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(ndp src/session.cpp src/functional_backend.cpp)
target_include_directories(ndp PUBLIC include)
target_compile_features(ndp PUBLIC cxx_std_20)
target_link_libraries(ndp PRIVATE Threads::Threads)
target_compile_options(ndp PRIVATE -Wall -Wextra -Wpedantic)
add_executable(host_roundtrip examples/host_roundtrip.cpp)
target_link_libraries(host_roundtrip PRIVATE ndp)
include(CTest)
if(BUILD_TESTING)
  add_executable(ndp_tests tests/session_test.cpp)
  target_link_libraries(ndp_tests PRIVATE ndp)
  add_test(NAME session_contracts COMMAND ndp_tests)
  add_executable(ndp_async_tests tests/async_test.cpp)
  target_link_libraries(ndp_async_tests PRIVATE ndp)
  add_test(NAME async_contracts COMMAND ndp_async_tests)
  set_tests_properties(session_contracts async_contracts PROPERTIES TIMEOUT 20)
endif()
